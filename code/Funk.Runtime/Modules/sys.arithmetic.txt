#	Regroup numerical types under on type
#	Numerical types will be supported for all arithmetic operations
type numerical = integer | real | rational;

#	Reduce a rational number / fraction
private native let reducing(rational r);

#	Conversion to real
native let roReal(integer a);

let roReal(rational a)
{
	toReal(a.numerator) / toReal(a.denominator);
}

let roReal(real a)
{
	a;
}

#	Additions
#	Integers only
native let add(integer a, integer b);

#	Integers and rationals
let add(rational a, integer b)
{
	reducing(rational(a.numerator+b*a.denominator, a.denominator));
}

let add(integer a, rational b)
{
	add(b, a);
}

let add(rational a, rational b)
{
	reducing(
		rational(
			a.numerator*b.denominator+b.numerator*a.denominator,
			a.denominator*b.denominator));
}

#	Integers, rationals & reals
native let add(real a, real b);

let add(real a, integer b)
{
	a + toReal(b);
}

let add(integer a, real b)
{
	add(b, a);
}

let add(real a, rational b)
{
	a + toReal(b);
}

let add(rational a, real b)
{
	add(b, a);
}

#	Negation
native let negation(integer a);

let negation(rational a)
{
	rational(-a.numerator, a.denominator);
}

native let negation(real a);

#	Multiplication

native let product(integer a, integer b);

#	Integers and rationals

{
	reducing(rational(a.numerator*b, a.denominator));

let product(integer a, rational b)
{
	product(b, a);

let product(rational a, rational b)
{
	reducing(rational(a.numerator*b.numerator, a.denominator*b.denominator));

let product(real a, integer b)

#	Division
#	Integers only
native let division(integer a, integer b);

#	Integers and rationals
let div(rational a, integer b)
{
	reducing(rational(a.numerator, a.denominator*b));
}

let div(integer a, rational b)
{
	reducing(rational(a*b.denominator, b.numerator));
}

let div(rational a, rational b)
{
	reducing(rational(a.numerator*b.denominator, a.denominator*b.numerator));
}

#	Integers, rationals & reals
native let div(real a, real b);

let div(real a, integer b)
{
	a / toReal(b);
}

let div(integer a, real b)
{
	toReal(a) / b;
}

let div(real a, rational b)
{
	a / toReal(b);
}

let div(rational a, real b)
{
	toReal(a) / b;
}

##	Power
native let pow(real a, real b);

let pow(integer | rational | real a, integer | rational b)
{
	pow(toReal(a), toReal(b));
}

let pow(integer | rational a, integer | rational | real b)
{
	pow(toReal(a), toReal(b));
}
