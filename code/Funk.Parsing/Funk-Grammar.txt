#	Comments & interleaves
rule(interleave=false) comment = "//" (. - ("\r" | "\n"))*;
rule(interleave=false) blockSpacer = " " | "\t" | "\r";
rule(interleave=false) padding = blockSpacer | "\n" | comment;
interleave = padding;

#	tokens
rule(interleave=false) letter = ("a".."z" | "A".."Z");
rule(interleave=false) number = ("0".."9");
rule(interleave=false) identifier = letter (letter | number)* => text;
rule(interleave=false) integer = "-"? number+ => integer(text);
rule(interleave=false) real = "-"? number* "." number+ => float(text);
rule(interleave=false) singleQuotableString = ((. - "'") | "\\'")* => text;
rule(interleave=false) doubleQuotableString = ((. - '"') | '\\"')* => text;
rule(interleave=false) singleQuotedString = "'" q:singleQuotableString "'" => q;
rule(interleave=false) doubleQuotedString = '"' q:doubleQuotableString '"' => q;
rule(interleave=false) quotedString = doubleQuotedString | singleQuotedString;

#	Litterals
rule primitive = integer:integer | real:real | string:quotedString;

#	Binary operators
rule binaryOperator = ("+" => "add")
	| ("-" => "substract")
	| ("*" => "product")
	| ("/" => "division")
	| ("^" => "power")
	| ("==" => "equals")
	| ("!=" => "notEquals")
	| ("<=" => "lesserOrEquals")
	| ("<" => "lesser")
	| (">=" => "greaterOrEquals")
	| (">" => "greater")
	| ("&&" => "and")
	| ("||" => "or");
rule binaryOperation = left:nonBinaryExpression operator:binaryOperator right:expression;

#	Import statements
rule importStatement = "import" id:(identifier ("." identifier)* => text) ";" => {"module":id};

#	Function declaration
rule parameterDeclaration = identifier => { "id" : text };
rule parameterDeclarationList = head:parameterDeclaration tail:("," pd:parameterDeclaration => pd)*
	=> prepend(head, tail);
rule parameterDeclarationParenthesis = "(" el:parameterDeclarationList? ")"
	=> coalesce([firstOrNull(el), []]);
rule succinctFunctionBody = e:expression ";" =>
{
	"importStatements" :[],
	"functionDeclarations" : [],
	"functionEvaluation" : e
};
rule fullFunctionBody = "{" is:importStatement* fds:functionDeclarationStatement* fes:expression "}" =>
{
	"importStatements" :is,
	"functionDeclarations" : fds,
	"functionEvaluation" : fes
};
rule functionBody = succinctFunctionBody | fullFunctionBody;
rule functionDeclarationStatement = "let" name:identifier
	parameters:parameterDeclarationParenthesis? "=" body:functionBody
	=> {
		"name" : name,
		"parameters" : coalesce([firstOrNull(parameters), []]),
		"body" : body
	};

#	Function invoke
rule expressionList = head:expression tail:("," e:expression => e)*
	=> prepend(head, tail);
rule functionInvoke = name:identifier parameters:("(" el:expressionList? ")" => firstOrNull(el))?
	=>
	{
		"name" : name,
		"parameters" : coalesce([firstOrNull(parameters), []])
	};

#	Parameter access
rule parameterAccess = expression:nonBinaryExpression names:("." name:identifier => name)+;

#	if / conditionals
rule scopedExpression = "{" "return" e:expression ";" "}" => e;
rule elseIf = "else" "if" "(" condition:expression ")" thenExpression:scopedExpression;
rule chainedIfElse = leadIf:("if" "(" condition:expression ")" thenExpression:scopedExpression)
	elseIfs:elseIf* "else" elseExpression:scopedExpression
	=> { "ifThens" : prepend(leadIf, elseIfs), "elseExpression" : elseExpression };
rule ternaryIf = ifThen:("if" "(" condition:expression "," thenExpression:expression) ","
	elseExpression:expression ")"
	=> { "ifThens" : [ifThen], "elseExpression" : elseExpression };
rule if = ternaryIf | chainedIfElse;

#	Expression
rule parenthesis = "(" e:expression ")" => e;
rule nonBinaryExpression = if:if
	| parenthesis:parenthesis
	| primitive:primitive
	| functionInvoke:functionInvoke;
rule expression = nonBinaryExpression
	| (parameterAccess => { "parameterAccess" : defaultOutput })
	| (binaryOperation => { "binaryOperation" : defaultOutput });

#	Function evaluation statement
rule functionEvaluationStatement = expression => { "text" : text, "functionEvaluation" : defaultOutput };

#	main rules
rule adHocScript = is:importStatement* fds:functionDeclarationStatement*
	fes:functionEvaluationStatement? =>
	{
		"importStatements" : is,
		"functionDeclarations" : fds,
		"functionEvaluation" : firstOrNull(fes)
	};
rule main = adHocScript;
